### 問題 1: 日期解析在本週篩選中可能失敗

**位置**: `public/js/pages/learning-record-upload.js` - 第 11703-11714 行

**問題代碼**:
```javascript
var dateParts = record.date.split(' ');
var dateStr = dateParts[0];
var recordDate = new Date(dateStr);

if (isNaN(recordDate.getTime())) return false;
```

**問題描述**:
- `record.date` 格式可能為 `"2025-11-05 四足獸"` 或 `"2025-11-05"`
- 如果日期格式不標準（例如：`"2025-11-5"`、`"2025/11/05"`），`new Date()` 解析可能失敗
- 在某些瀏覽器中，`new Date("YYYY-MM-DD")` 會被解析為 UTC 時間，導致時區問題

**影響**:
- 本週篩選可能不準確
- 某些記錄可能被錯誤地排除

**建議修復**:
```javascript
// 🔥 改進版：更健壯的日期解析
var dateParts = record.date.split(' ');
var dateStr = dateParts[0];

// 確保日期格式為 YYYY-MM-DD（補零）
var dateMatch = dateStr.match(/(\d{4})-(\d{1,2})-(\d{1,2})/);
if (!dateMatch) return false;

var year = dateMatch[1];
var month = dateMatch[2].padStart(2, '0');
var day = dateMatch[3].padStart(2, '0');
var normalizedDateStr = `${year}-${month}-${day}`;

// 使用明確的本地時區解析
var recordDate = new Date(normalizedDateStr + 'T00:00:00');

if (isNaN(recordDate.getTime())) return false;
```

### 問題 2: 講師/課別篩選可能遺漏部分記錄

**位置**: `public/js/pages/learning-record-upload.js` - 第 11734-11755 行

**問題代碼**:
```javascript
records = records.filter(function(record, idx) {
  var recordTeacher = extractTeacherFromRecord(record, idx < 5);
  var recordCourseType = extractCourseTypeFromTitle(record.course);
  
  var teacherMatch = !teacher || (recordTeacher && recordTeacher === teacher);
  var courseTypeMatch = !courseType || (recordCourseType && recordCourseType === courseType);
  
  return teacherMatch && courseTypeMatch;
});
```

**問題描述**:
1. **大小寫敏感問題**: 
   - `recordTeacher === teacher` 是嚴格相等比較
   - 如果後端返回 `"Agnes"` 但篩選條件是 `"AGNES"`，將不匹配
   
2. **缺少空值保護**:
   - 如果 `extractTeacherFromRecord` 返回空字串 `""`，而 `teacher` 是某個具體值
   - 條件 `recordTeacher && recordTeacher === teacher` 會是 `false`（正確）
   - 但如果 `teacher` 是空字串，所有記錄都會通過（可能不符合預期）

**建議修復**:
```javascript
records = records.filter(function(record, idx) {
  var recordTeacher = extractTeacherFromRecord(record, idx < 5);
  var recordCourseType = extractCourseTypeFromTitle(record.course);
  
  // 🔥 改進：大小寫不敏感比較
  var teacherMatch = !teacher || (
    recordTeacher && 
    recordTeacher.toUpperCase() === teacher.toUpperCase()
  );
  
  var courseTypeMatch = !courseType || (
    recordCourseType && 
    recordCourseType.toUpperCase() === courseType.toUpperCase()
  );
  
  return teacherMatch && courseTypeMatch;
});
```

**位置**: `public/js/pages/learning-record-upload.js` - 第 11545-11563 行

  var teacher = teacherNames[i];

  var regex = new RegExp('\\b' + teacherUpper + '\\b');
  if (regex.test(titleUpper)) {

### 問題 7: 記錄排序邏輯簡單

**位置**: `public/js/pages/learning-record-upload.js` - 第 11808-11810 行

**問題代碼**:
```javascript
var groupedArray = Object.values(groupedRecords).sort(function(a, b) {
  return b.date.localeCompare(a.date);
});
```

**問題描述**:
- **僅按日期排序**: 如果同一天有多堂課，順序可能不符合預期
- **日期格式依賴**: `localeCompare` 對 `"2025-11-05 主題"` 和 `"2025-11-05"` 的比較可能不一致
- **缺少次要排序**: 同一天的課程應該按時間排序

**建議改進**:
```javascript
var groupedArray = Object.values(groupedRecords).sort(function(a, b) {
  // 提取純日期部分
  var dateA = a.date.split(' ')[0];
  var dateB = b.date.split(' ')[0];
  
  // 首先按日期降序（最新在前）
  var dateCompare = dateB.localeCompare(dateA);
  if (dateCompare !== 0) return dateCompare;
  
  // 日期相同，按課程時間排序
  var timeA = extractTimeFromCourse(a.course) || '00:00';
  var timeB = extractTimeFromCourse(b.course) || '00:00';
  return timeA.localeCompare(timeB);
});

// 輔助函數：從課程名稱提取時間
function extractTimeFromCourse(courseName) {
  var match = courseName.match(/(\d{1,2}:\d{2})/);
  return match ? match[1] : null;
}
```

  if (typeof unsafe !== 'string') return unsafe;
  
  return unsafe
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}
