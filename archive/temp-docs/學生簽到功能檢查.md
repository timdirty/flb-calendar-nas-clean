   ```javascript
   async function updateGoogleSheets(teacher, course, time, studentName, status, today) {
       const payload = {
           action: "update",
           name: studentName,  // 學生姓名
           date: today,
           present: status === 'present' ? true : false,
           teacher: teacher,
           course: course,
           period: time
       };
       
       // 通過後端代理發送到 Google Sheets
       const response = await fetch('/api/proxy/google-sheets', {
           method: 'POST',
           body: JSON.stringify({
               action: 'updateAttendance',
               googleSheetsUrl: "https://script.google.com/macros/s/AKfycbxfj5fwNIc8ncbqkOm763yo6o06wYPHm2nbfd_1yLkHlakoS9FtYfYJhvGCaiAYh_vjIQ/dev",
               payload: payload
           })
       });
   }
   ```
