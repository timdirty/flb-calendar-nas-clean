## 2. 後端 API 開發

**檔案**: `server.js`

### 2.1 新增 API endpoint: `/api/send-temporary-student-notification`

- 方法：POST
- 接收參數：
  - `studentId`: 臨時學生 ID
  - `type`: 'makeup' 或 'trial'
- 功能流程：

  1. 從 `temporary_students.json` 讀取學生資料
  2. 驗證學生是否存在且有 userId
  3. 從 Synology Calendar 獲取課程資料（講師姓名、課程詳情）
  4. 根據 type 選擇對應的 Flex Message 範本
  5. 替換範本變數：

     - `{studentName}`: 學生姓名
     - `{courseName}`: 課程名稱
     - `{courseDate}`: 日期
     - `{weekday}`: 星期
     - `{courseTime}`: 時段
     - `{originalPeriod}`: 原本時段（僅補課）
     - `{teacherName}`: 講師名稱
     - `{note}`: 備註

  1. 使用 `notificationManager.sendLineMessage()` 發送
  2. 回傳結果：`{ success: true/false, message: '...' }`

### 2.2 修改 `/api/temporary-students` (POST)

在新增臨時學生成功後，加入自動通知邏輯：

- 檢查是否有 userId
- 如果有，自動調用發送通知功能
- 回傳結果中包含 `notificationSent: true/false`

**JavaScript 函數**（替換現有的 `showAlert`）：

```javascript
// 🔥 新的 Toast 系統
function showToast(message, type = 'info', duration = 3000) {
    const container = document.getElementById('toastContainer');
    
    // 圖標映射
    const icons = {
        success: '✅',
        error: '❌',
        warning: '⚠️',
        info: 'ℹ️'
    };
    
    // 創建 toast 元素
    const toast = document.createElement('div');
    toast.className = `toast toast-${type}`;
    toast.innerHTML = `
        <span class="toast-icon">${icons[type] || icons.info}</span>
        <span class="toast-message">${message}</span>
        <button class="toast-close" onclick="this.parentElement.remove()">×</button>
    `;
    
    container.appendChild(toast);
    
    // 自動移除
    if (duration > 0) {
        setTimeout(() => {
            toast.classList.add('removing');
            setTimeout(() => toast.remove(), 300);
        }, duration);
    }
}

// 保留舊函數名稱以維持相容性
function showAlert(message, type = 'info', duration = 3000) {
    showToast(message, type, duration);
}
```

### 3.3 新增手動發送通知函數

```javascript
// 🔥 手動發送補課/體驗通知給家長
async function sendNotificationToParent(studentId, type) {
    try {
        showToast('🔄 正在發送通知...', 'info', 0);
        
        const response = await fetch('/api/send-temporary-student-notification', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ studentId, type })
        });
        
        const result = await response.json();
        
        // 移除載入中的 toast
        document.querySelectorAll('.toast').forEach(t => t.remove());
        
        if (result.success) {
            const typeLabel = type === 'makeup' ? '補課設定' : '體驗課程建立';
            showToast(`✅ ${typeLabel}通知已發送！`, 'success');
        } else {
            showToast(`❌ 發送失敗：${result.message}`, 'error');
        }
    } catch (error) {
        console.error('發送通知失敗:', error);
        showToast('❌ 發送通知失敗：' + error.message, 'error');
    }
}
```
