### 2.1 建立新的媒體處理模組

建立 `utils/media-processor.js`：

```javascript
const sharp = require('sharp');
const path = require('path');
const fs = require('fs').promises;
const PQueue = require('p-queue').default;

// 處理隊列：限制並發數，避免 CPU 過載
const imageQueue = new PQueue({ concurrency: 2 });
const videoQueue = new PQueue({ concurrency: 1 });

/**
 * 生成多尺寸縮圖（WebP 格式）
 * @param {string} originalPath - 原始圖片路徑
 * @param {string} outputDir - 輸出目錄
 * @returns {Promise<Object>} 包含各尺寸縮圖路徑
 */
async function generateThumbnails(originalPath, outputDir) {
  const sizes = [
    { name: 'thumb', width: 200, height: 200 },
    { name: 'medium', width: 800, height: 800 },
    { name: 'large', width: 1920, height: 1920 }
  ];
  
  const results = {};
  await fs.mkdir(outputDir, { recursive: true });
  
  for (const size of sizes) {
    const filename = `${path.basename(originalPath, path.extname(originalPath))}_${size.name}.webp`;
    const outputPath = path.join(outputDir, filename);
    
    await sharp(originalPath)
      .resize(size.width, size.height, { fit: 'inside', withoutEnlargement: true })
      .webp({ quality: 85 })
      .toFile(outputPath);
    
    results[size.name] = outputPath;
  }
  
  return results;
}

// 導出隊列化版本
module.exports = {
  generateThumbnails: (original, output) => imageQueue.add(() => generateThumbnails(original, output)),
  imageQueue,
  videoQueue
};
```

### 2.2 建立分片上傳 API

在 `server.js` 新增分片上傳端點（位於現有 `/api/learning-records/upload` 之前）：

```javascript
// ==================== 分片上傳 API ====================
const chunkUploads = new Map(); // 儲存分片上傳狀態

// 初始化分片上傳
app.post('/api/learning-records/upload/init', (req, res) => {
  const { filename, fileSize, fileType, chunkSize } = req.body;
  const uploadId = require('uuid').v4();
  const totalChunks = Math.ceil(fileSize / chunkSize);
  
  chunkUploads.set(uploadId, {
    filename,
    fileSize,
    fileType,
    totalChunks,
    receivedChunks: new Set(),
    createdAt: Date.now()
  });
  
  res.json({ success: true, uploadId, totalChunks });
});

// 接收單一分片
app.post('/api/learning-records/upload/chunk', upload.single('chunk'), (req, res) => {
  const { uploadId, chunkIndex } = req.body;
  const uploadState = chunkUploads.get(uploadId);
  
  if (!uploadState) {
    return res.status(404).json({ success: false, message: '上傳會話不存在' });
  }
  
  const chunkDir = path.join(__dirname, 'data', 'upload-chunks', uploadId);
  fs.mkdirSync(chunkDir, { recursive: true });
  
  const chunkPath = path.join(chunkDir, `chunk_${chunkIndex}`);
  fs.renameSync(req.file.path, chunkPath);
  
  uploadState.receivedChunks.add(parseInt(chunkIndex));
  
  res.json({ 
    success: true, 
    receivedChunks: uploadState.receivedChunks.size,
    totalChunks: uploadState.totalChunks
  });
});

// 完成分片上傳（合併檔案）
app.post('/api/learning-records/upload/complete', async (req, res) => {
  const { uploadId, targetPath, metadata } = req.body;
  const uploadState = chunkUploads.get(uploadId);
  
  if (!uploadState || uploadState.receivedChunks.size !== uploadState.totalChunks) {
    return res.status(400).json({ success: false, message: '分片不完整' });
  }
  
  try {
    const chunkDir = path.join(__dirname, 'data', 'upload-chunks', uploadId);
    const finalPath = path.join(__dirname, 'data', 'learning-portfolio', targetPath);
    await fs.mkdir(path.dirname(finalPath), { recursive: true });
    
    const writeStream = fs.createWriteStream(finalPath);
    for (let i = 0; i < uploadState.totalChunks; i++) {
      const chunkPath = path.join(chunkDir, `chunk_${i}`);
      const chunkBuffer = await fs.readFile(chunkPath);
      writeStream.write(chunkBuffer);
    }
    writeStream.end();
    
    // 清理分片
    await fs.rm(chunkDir, { recursive: true, force: true });
    chunkUploads.delete(uploadId);
    
    // 異步處理縮圖（不阻塞回應）
    if (uploadState.fileType.startsWith('image/')) {
      const mediaProcessor = require('./utils/media-processor');
      const thumbDir = path.join(path.dirname(finalPath), 'thumbnails');
      mediaProcessor.generateThumbnails(finalPath, thumbDir).catch(err => {
        console.error('❌ 縮圖生成失敗:', err);
      });
    }
    
    res.json({ success: true, path: finalPath });
  } catch (error) {
    console.error('❌ 合併檔案失敗:', error);
    res.status(500).json({ success: false, message: error.message });
  }
});
```

## 第三階段：前端上傳優化

### 3.1 建立分片上傳工具模組

建立 `public/js/modules/chunked-uploader.js`：

```javascript
(function(global) {
  'use strict';
  
  const CHUNK_SIZE = 5 * 1024 * 1024; // 5MB
  
  /**
   * 分片上傳單一檔案
   * @param {File} file - 檔案物件
   * @param {Function} onProgress - 進度回調 (percent)
   * @returns {Promise<Object>} 上傳結果
   */
  async function uploadFileChunked(file, onProgress) {
    // 1. 初始化上傳
    const initRes = await fetch('/api/learning-records/upload/init', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        filename: file.name,
        fileSize: file.size,
        fileType: file.type,
        chunkSize: CHUNK_SIZE
      })
    }).then(r => r.json());
    
    if (!initRes.success) throw new Error('初始化上傳失敗');
    
    const { uploadId, totalChunks } = initRes;
    
    // 2. 並行上傳分片
    const uploadChunk = async (index) => {
      const start = index * CHUNK_SIZE;
      const end = Math.min(start + CHUNK_SIZE, file.size);
      const chunk = file.slice(start, end);
      
      const formData = new FormData();
      formData.append('chunk', chunk);
      formData.append('uploadId', uploadId);
      formData.append('chunkIndex', index);
      
      const res = await fetch('/api/learning-records/upload/chunk', {
        method: 'POST',
        body: formData
      }).then(r => r.json());
      
      if (res.success) {
        onProgress(Math.round((res.receivedChunks / totalChunks) * 100));
      }
      
      return res;
    };
    
    // 使用 Promise.all 並行上傳（限制並發數）
    const concurrency = 3;
    for (let i = 0; i < totalChunks; i += concurrency) {
      const batch = Array.from({ length: Math.min(concurrency, totalChunks - i) }, (_, j) => uploadChunk(i + j));
      await Promise.all(batch);
    }
    
    // 3. 完成上傳
    const completeRes = await fetch('/api/learning-records/upload/complete', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ uploadId })
    }).then(r => r.json());
    
    return completeRes;
  }
  
  global.ChunkedUploader = { uploadFileChunked };
})(window);
```

### 6.1 功能測試清單

- 小檔案上傳（< 10MB）：驗證原有流程不受影響

- 批次上傳：同時上傳 10+ 個檔案，驗證隊列處理

## 技術決策說明

### 為何不使用 MinIO？

現階段保留 NAS 檔案系統，原因：

1. 現有系統已穩定運行，風險較低
2. MinIO 需額外容器與網路配置
3. 檔案數量級（< 10萬）尚可用檔案系統管理
4. 未來可漸進遷移：原始檔保留 NAS、縮圖遷移至 MinIO

### 為何不使用 RabbitMQ？

使用 Node.js p-queue 函式庫，原因：

1. 單機部署足夠應付當前負載（50 人並發）
2. 減少系統複雜度與維護成本
3. 內建記憶體隊列延遲低、效能佳
4. 重啟後遺失任務可接受（縮圖可重新生成）

### Sharp vs ImageMagick

選用 Sharp，原因：

1. 效能：處理速度快 4-8 倍
2. 記憶體：佔用僅 ImageMagick 的 1/10
3. 安裝：純 npm 依賴，無需系統套件
4. 格式：原生支援 WebP/AVIF
