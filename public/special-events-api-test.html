📚 JavaScript 使用範例

                
                

                    
// 1. 取得所有特殊事件類型配置
async function getSpecialEventTypes() {
    const response = await fetch('/api/special-event-types');
    const data = await response.json();
    console.log('特殊事件配置:', data.data);
    return data.data;
}

// 2. 偵測單一事件
async function checkEvent(title) {
    const response = await fetch('/api/detect-special-event', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ title })
    });
    const data = await response.json();
    console.log(`"${title}" 偵測結果:`, data.specialEventType);
    return data;
}

// 3. 批量偵測
async function checkBatchEvents(titles) {
    const response = await fetch('/api/detect-batch-events', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ titles })
    });
    const data = await response.json();
    console.log('批量偵測結果:', data.results);
    return data;
}

// 使用範例
(async () => {
    // 取得配置
    const config = await getSpecialEventTypes();
    
    // 單一偵測
    await checkEvent('Python 停課通知');
    
    // 批量偵測
    const titles = ['Python 停課', '體驗課程', '正常課程'];
    await checkBatchEvents(titles);
})();

                

            


            

                
🐍 Python 使用範例

                
                

                    
import requests
import json

# API 基礎 URL
BASE_URL = 'http://your-domain.com'

# 1. 取得特殊事件配置
def get_special_event_types():
    response = requests.get(f'{BASE_URL}/api/special-event-types')
    return response.json()['data']

# 2. 偵測單一事件
def detect_event(title):
    response = requests.post(
        f'{BASE_URL}/api/detect-special-event',
        json={'title': title}
    )
    return response.json()

# 3. 批量偵測
def detect_batch_events(titles):
    response = requests.post(
        f'{BASE_URL}/api/detect-batch-events',
        json={'titles': titles}
    )
    return response.json()

# 使用範例
if __name__ == '__main__':
    # 取得配置
    config = get_special_event_types()
    print('特殊事件配置:', json.dumps(config, ensure_ascii=False, indent=2))
    
    # 單一偵測
    result = detect_event('Python 停課通知')
    print(f"偵測結果: {result['specialEventType']}")
    
    # 批量偵測
    titles = ['Python 停課', '體驗課程', '正常課程']
    batch_result = detect_batch_events(titles)
    for item in batch_result['results']:
        print(f"{item['title']}: {item['specialEventType'] or '正常'}")
