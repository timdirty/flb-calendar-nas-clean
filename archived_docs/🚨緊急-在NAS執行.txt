echo "宿主機 API 數: $HOST_API" && \
echo "容器內 API 數: $CONTAINER_API" && \
if [ "$HOST_API" != "$CONTAINER_API" ]; then \
  echo "⚠️ 版本不一致！需要重建容器..." && \
  sudo docker-compose down && \
  sudo docker-compose build --no-cache && \
  sudo docker-compose up -d && \
  sleep 15 && \
  echo "✅ 容器已重建" && \
  sudo docker-compose ps && \
  echo "" && \
  echo "🧪 測試 API:" && \
  curl -s https://calendar.funlearnbar.synology.me/api/line-config | head -20; \
else \
  echo "✅ 版本一致，只需重啟..." && \
  sudo docker-compose restart && \
  sleep 10 && \
  sudo docker-compose ps; \
fi
