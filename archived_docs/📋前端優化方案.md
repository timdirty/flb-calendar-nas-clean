### 1. 創建統一的液態玻璃通知系統

```javascript
// 綁定專用通知（液態玻璃效果）
function showBindingNotification(message, type = 'success', options = {}) {
    const notification = document.createElement('div');
    const isMobile = window.innerWidth <= 768;
    
    // 顏色方案
    const colorSchemes = {
        success: {
            gradient: 'linear-gradient(135deg, rgba(16, 185, 129, 0.95), rgba(5, 150, 105, 0.9))',
            glow: 'rgba(16, 185, 129, 0.5)',
            icon: '✓'
        },
        processing: {
            gradient: 'linear-gradient(135deg, rgba(59, 130, 246, 0.95), rgba(37, 99, 235, 0.9))',
            glow: 'rgba(59, 130, 246, 0.5)',
            icon: '⟳'
        },
        warning: {
            gradient: 'linear-gradient(135deg, rgba(245, 158, 11, 0.95), rgba(217, 119, 6, 0.9))',
            glow: 'rgba(245, 158, 11, 0.5)',
            icon: '⚠'
        },
        error: {
            gradient: 'linear-gradient(135deg, rgba(239, 68, 68, 0.95), rgba(220, 38, 38, 0.9))',
            glow: 'rgba(239, 68, 68, 0.5)',
            icon: '✕'
        }
    };
    
    const scheme = colorSchemes[type] || colorSchemes.success;
    
    notification.className = 'binding-notification';
    notification.style.cssText = `
        position: fixed;
        top: ${isMobile ? '80px' : '100px'};
        left: 50%;
        transform: translateX(-50%) translateY(-20px);
        opacity: 0;
        z-index: 10000;
        
        /* 液態玻璃效果 */
        background: ${scheme.gradient};
        backdrop-filter: blur(30px) saturate(180%);
        -webkit-backdrop-filter: blur(30px) saturate(180%);
        
        /* 邊框和陰影 */
        border: 1px solid rgba(255, 255, 255, 0.3);
        border-radius: ${isMobile ? '20px' : '25px'};
        box-shadow: 
            0 20px 60px ${scheme.glow},
            0 0 0 1px rgba(255, 255, 255, 0.1) inset,
            0 10px 40px rgba(0, 0, 0, 0.3);
        
        /* 內容 */
        padding: ${isMobile ? '16px 24px' : '20px 32px'};
        color: white;
        font-size: ${isMobile ? '15px' : '16px'};
        font-weight: 600;
        text-align: center;
        max-width: ${isMobile ? '85vw' : '500px'};
        
        /* 性能優化 */
        will-change: transform, opacity;
        contain: layout style paint;
        transform: translateZ(0);
    `;
    
    // 圖標 + 文字
    notification.innerHTML = `
        <div style="display: flex; align-items: center; justify-content: center; gap: 12px;">
            <div style="
                width: ${isMobile ? '32px' : '40px'};
                height: ${isMobile ? '32px' : '40px'};
                border-radius: 50%;
                background: rgba(255, 255, 255, 0.2);
                display: flex;
                align-items: center;
                justify-content: center;
                font-size: ${isMobile ? '18px' : '22px'};
                ${type === 'processing' ? 'animation: rotate 1.5s linear infinite;' : ''}
            ">${scheme.icon}</div>
            <div>${message}</div>
        </div>
    `;
    
    // 添加旋轉動畫樣式
    if (type === 'processing' && !document.getElementById('rotate-animation')) {
        const style = document.createElement('style');
        style.id = 'rotate-animation';
        style.textContent = `
            @keyframes rotate {
                from { transform: rotate(0deg); }
                to { transform: rotate(360deg); }
            }
        `;
        document.head.appendChild(style);
    }
    
    document.body.appendChild(notification);
    
    // 顯示動畫
    requestAnimationFrame(() => {
        notification.style.transform = 'translateX(-50%) translateY(0)';
        notification.style.opacity = '1';
        notification.style.transition = 'all 0.5s cubic-bezier(0.34, 1.56, 0.64, 1)';
    });
    
    // 自動消失（如果不是持續顯示）
    if (!options.persist) {
        setTimeout(() => {
            notification.style.transform = 'translateX(-50%) translateY(-20px)';
            notification.style.opacity = '0';
            setTimeout(() => notification.remove(), 500);
        }, options.duration || 3000);
    }
    
    return notification; // 返回元素以便手動控制
}
```

### 2. 手機端性能優化

```javascript
// 性能優化管理器（針對手機端）
const MobileOptimizer = {
    // 檢測設備
    isMobile: /Android|iPhone|iPad|iPod/i.test(navigator.userAgent),
    isLowEnd: (() => {
        const memory = navigator.deviceMemory;
        const cores = navigator.hardwareConcurrency;
        return memory && memory < 4 || cores && cores < 4;
    })(),
    
    // 初始化優化
    init() {
        if (!this.isMobile) return;
        
        console.log('🔧 手機端優化啟動');
        
        // 1. 減少動畫複雜度
        if (this.isLowEnd) {
            this.simplifyAnimations();
        }
        
        // 2. 使用 Intersection Observer 延遲載入
        this.setupLazyLoad();
        
        // 3. 優化滾動性能
        this.optimizeScroll();
        
        // 4. 減少重繪
        this.reduceRepaints();
    },
    
    // 簡化動畫
    simplifyAnimations() {
        const style = document.createElement('style');
        style.textContent = `
            * {
                animation-duration: 0.3s !important;
                transition-duration: 0.3s !important;
            }
            .blur-effect {
                backdrop-filter: none !important;
                -webkit-backdrop-filter: none !important;
            }
        `;
        document.head.appendChild(style);
    },
    
    // 延遲載入
    setupLazyLoad() {
        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.classList.add('loaded');
                }
            });
        }, {
            rootMargin: '50px'
        });
        
        document.querySelectorAll('.lazy-load').forEach(el => {
            observer.observe(el);
        });
    },
    
    // 優化滾動
    optimizeScroll() {
        let scrollTimer;
        const scrollContainer = document.getElementById('eventsContainer');
        
        if (scrollContainer) {
            scrollContainer.addEventListener('scroll', () => {
                clearTimeout(scrollTimer);
                scrollContainer.classList.add('is-scrolling');
                
                scrollTimer = setTimeout(() => {
                    scrollContainer.classList.remove('is-scrolling');
                }, 150);
            }, { passive: true });
        }
    },
    
    // 減少重繪
    reduceRepaints() {
        // 使用 DocumentFragment 批量 DOM 操作
        this.batchDOMUpdates = (updates) => {
            const fragment = document.createDocumentFragment();
            updates.forEach(update => update(fragment));
            return fragment;
        };
    }
};

// 在頁面載入時啟動
document.addEventListener('DOMContentLoaded', () => {
    MobileOptimizer.init();
});
```

### 3. 載入邏輯確認

當前載入流程：
```
1. 頁面載入
   ↓
2. 檢查 localStorage 快取
   - 如果有快取且未過期 → 直接使用
   - 如果快取為空（0 events）→ 忽略，從後端獲取
   ↓
3. 檢查後端快取狀態（isReady）
   - 如果未就緒 → 等待就緒（最多 40 秒）
   - 如果已就緒 → 直接請求
   ↓
4. 請求 /api/events
   - timeout: 35 秒（配合後端 30 秒等待）
   - retry: 2 次
   ↓
5. 處理事件資料
   - populateInstructorDropdown()（等待完成）
   - 自動比對講師
   ↓
6. 完成初始化
```
