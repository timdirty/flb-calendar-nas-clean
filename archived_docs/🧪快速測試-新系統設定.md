## 🚀 一鍵測試腳本

將以下腳本儲存為 `test-system-settings.sh`:

```bash
#!/bin/bash

echo "=========================================="
echo "🧪 測試新系統設定頁面"
echo "=========================================="
echo ""

API_BASE="http://localhost:3000"
PASSED=0
FAILED=0

test_api() {
    local name=$1
    local method=$2
    local endpoint=$3
    local data=$4
    
    echo -n "測試: $name ... "
    
    if [ "$method" = "GET" ]; then
        RESPONSE=$(curl -s -w "\n%{http_code}" "$API_BASE$endpoint")
    else
        RESPONSE=$(curl -s -w "\n%{http_code}" -X POST "$API_BASE$endpoint" \
            -H "Content-Type: application/json" -d "$data")
    fi
    
    HTTP_CODE=$(echo "$RESPONSE" | tail -n1)
    
    if [ "$HTTP_CODE" = "200" ] || [ "$HTTP_CODE" = "400" ]; then
        echo "✅ 通過 (HTTP $HTTP_CODE)"
        PASSED=$((PASSED + 1))
    else
        echo "❌ 失敗 (HTTP $HTTP_CODE)"
        FAILED=$((FAILED + 1))
    fi
}

# 執行所有測試
test_api "系統健康檢查" "GET" "/api/health"
test_api "取得管理員資訊" "GET" "/api/admin/info"
test_api "取得LINE配置" "GET" "/api/line-config"
test_api "取得系統設定" "GET" "/api/system-settings"
test_api "取得排程器狀態" "GET" "/api/reminder-scheduler/status"
test_api "測試行事曆連線" "GET" "/api/events"

test_api "設定管理員" "POST" "/api/admin/set" '{"adminUserId":"U1234567890abcdef"}'
test_api "啟動排程器" "POST" "/api/reminder-scheduler/start" '{}'
test_api "停止排程器" "POST" "/api/reminder-scheduler/stop" '{}'

echo ""
echo "=========================================="
echo "測試結果總結"
echo "=========================================="
echo "✅ 通過: $PASSED"
echo "❌ 失敗: $FAILED"
echo ""

if [ $FAILED -eq 0 ]; then
    echo "🎉 所有測試通過！"
    exit 0
else
    echo "⚠️  部分測試失敗，請檢查日誌"
    exit 1
fi
```

**執行測試**:
```bash
chmod +x test-system-settings.sh
./test-system-settings.sh
```
