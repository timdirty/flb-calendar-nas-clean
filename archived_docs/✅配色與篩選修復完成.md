### 1. 講師顏色正確載入與套用

**問題根源：**
- `loadTeacherColors()` 函數使用的是 `/teacher_data.json` 路徑
- 但系統實際應該使用 `/api/teachers` API 端點
- 顏色載入失敗時沒有備用方案

**修復方案：**

```javascript
// 🔥 優先使用 API 載入，如果失敗則使用 JSON 文件作為備用
async function loadTeacherColors() {
    try {
        // 1. 優先嘗試從 API 載入
        let response = await fetch('/api/teachers', {
            method: 'GET',
            cache: 'no-cache'
        });
        let data = await response.json();
        
        // 2. 處理不同的 API 回應格式
        if (data.success && data.data && data.data.teachers) {
            data = data.data;
        } else if (data.teachers) {
            // 已經是正確格式
        }
        
        console.log('✅ 使用 API 載入講師顏色');
    } catch (apiError) {
        // 3. API 失敗時使用 JSON 文件作為備用
        response = await fetch('/teacher_data.json', {
            method: 'GET',
            cache: 'no-cache'
        });
        data = await response.json();
        console.log('✅ 使用 JSON 文件載入講師顏色');
    }
    
    // 4. 轉換為 { 講師名稱: 顏色 } 格式
    instructorColors = {};
    data.teachers.forEach(teacher => {
        if (teacher.name && teacher.color) {
            instructorColors[teacher.name] = teacher.color;
        }
    });
    
    console.log('✅ 講師顏色載入成功:', instructorColors);
    return true;
}
```

**預期效果：**
- ✅ 優先從 API 載入講師顏色
- ✅ API 失敗時自動使用 JSON 文件備用
- ✅ 正確解析不同的 API 回應格式
- ✅ 講師名稱底色正確顯示 teacher_data.json 中的顏色
- ✅ 課程卡片中的講師標籤正確套用自訂顏色

function renderTodayEvents() {

### 3. 配色面板根據篩選條件動態更新

**問題根源：**
- `renderColorLegend()` 函數直接顯示所有講師和課程類型
- 沒有考慮當前的篩選條件和視圖
- 導致配色面板顯示過多不相關的資訊

**修復方案：**

```javascript
function renderColorLegend() {
    // 🔥 獲取當前篩選條件
    const instructorFilter = document.getElementById('instructorSelect')?.value || '';
    const timeFilter = document.getElementById('timeFilter')?.value || '';
    const dateFilter = document.getElementById('dateFilter')?.value || '';

    // 🔥 獲取當前顯示的事件（根據篩選條件和視圖）
    let filteredEvents = allEvents.filter(event => {
        // 1. 基本驗證
        if (!event || !event.start || !event.end || !event.title || !event.instructor) 
            return false;

        // 2. 講師篩選
        if (instructorFilter && event.instructor !== instructorFilter) 
            return false;

        // 3. 時段篩選
        if (timeFilter) {
            const hour = new Date(event.start).getHours();
            let timeMatch = false;
            switch (timeFilter) {
                case 'morning': timeMatch = hour >= 6 && hour < 12; break;
                case 'afternoon': timeMatch = hour >= 12 && hour < 18; break;
                case 'evening': timeMatch = hour >= 18 && hour < 24; break;
            }
            if (!timeMatch) return false;
        }

        // 4. 指定日期篩選
        if (dateFilter) {
            const eventDate = new Date(event.start);
            const filterDate = new Date(dateFilter);
            if (eventDate.toDateString() !== filterDate.toDateString()) 
                return false;
        }

        // 5. 視圖篩選（今日/本週/本月/全部）
        const eventDate = new Date(event.start);
        const now = new Date();
        
        switch (currentView) {
            case '今日':
                return eventDate.toDateString() === now.toDateString();
            case '本週':
                const weekStart = getWeekStart(now);
                const weekEnd = getWeekEnd(weekStart);
                return eventDate >= weekStart && eventDate <= weekEnd;
            case '本月':
                return eventDate.getFullYear() === now.getFullYear() && 
                       eventDate.getMonth() === now.getMonth();
            case '全部':
            default:
                return true;
        }
    });

    // 🔥 提取當前顯示的課程類型和講師
    const displayedCourseTypes = new Set();
    const displayedInstructors = new Set();
    
    filteredEvents.forEach(event => {
        if (event.title) {
            const type = parseCourseType(event.title);
            if (type && type !== 'UNKNOWN') {
                displayedCourseTypes.add(type);
            }
        }
        if (event.instructor) {
            displayedInstructors.add(event.instructor);
        }
    });

    // 🔥 渲染課程類型配色（只顯示當前篩選出的類型）
    const courseLegend = document.getElementById('courseLegend');
    if (courseLegend && COURSE_TYPE_COLORS) {
        const filteredCourseTypes = Object.entries(COURSE_TYPE_COLORS)
            .filter(([type]) => displayedCourseTypes.has(type));
        
        if (filteredCourseTypes.length > 0) {
            courseLegend.innerHTML = filteredCourseTypes
                .map(([type, color]) => `
                    <div class="legend-item">
                        <span class="legend-dot" style="background: ${color};"></span>
                        <span>${type}</span>
                    </div>
                `).join('');
        } else {
            courseLegend.innerHTML = '<div class="legend-item" style="color: rgba(51, 51, 51, 0.5);">無課程類型</div>';
        }
    }
    
    // 🔥 渲染講師配色（只顯示當前篩選出的講師）
    const instructorLegend = document.getElementById('instructorLegend');
    if (instructorLegend && instructorColors) {
        const filteredInstructors = Object.entries(instructorColors)
            .filter(([name]) => displayedInstructors.has(name));
        
        if (filteredInstructors.length > 0) {
            instructorLegend.innerHTML = filteredInstructors
                .map(([name, color]) => `
                    <div class="legend-item">
                        <span class="legend-dot" style="background: ${color};"></span>
                        <span>${name}</span>
                    </div>
                `).join('');
        } else {
            instructorLegend.innerHTML = '<div class="legend-item" style="color: rgba(51, 51, 51, 0.5);">無講師</div>';
        }
    }

    console.log('🎨 配色面板已更新:', {
        課程類型: Array.from(displayedCourseTypes),
        講師: Array.from(displayedInstructors)
    });
}
```

**修復位置：** 第 9023-9175 行

**預期效果：**
- ✅ 選擇特定講師時，配色面板只顯示該講師
- ✅ 切換視圖（今日/本週/本月/全部）時，配色面板即時更新
- ✅ 應用時段篩選時，配色面板只顯示該時段涉及的講師和課程類型
- ✅ 沒有相關內容時顯示「無講師」或「無課程類型」
- ✅ 配色面板與課程列表完全同步
