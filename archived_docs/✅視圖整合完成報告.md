### 2. ✅ 創建統一的視圖切換函數

#### 新增 `updateSpecialEventsView()` 函數（第 8568-8605 行）

```javascript
// 🎯 整合視圖切換：根據原有視圖按鈕控制 Special Events Manager
function updateSpecialEventsView(view) {
    // 將視圖名稱映射到 Special Events Manager 的 range
    const rangeMap = {
        'today': 'today',
        'week': 'week',
        'month': 'month',
        'all': 'today' // 全部視圖時也顯示今日區塊，但隱藏日曆網格
    };
    
    const range = rangeMap[view] || 'today';
    specialEventsState.timeRange = range;
    
    console.log(`📅 Special Events Manager 狀態更新: ${range}`);
    
    // 檢測設備類型（手機/平板/桌機）
    const isDesktop = window.innerWidth > 768;
    
    // 控制日曆網格顯示
    const calendarSection = document.getElementById('calendarSection');
    if (range === 'today' || view === 'all') {
        // 今日視圖或全部視圖：隱藏日曆網格
        if (calendarSection) calendarSection.style.display = 'none';
    } else {
        // 本週/本月視圖：根據設備類型決定是否顯示
        if (calendarSection && isDesktop) {
            calendarSection.style.display = 'block';
            renderCalendarView();
            console.log(`📅 已渲染${range === 'week' ? '週' : '月'}日曆視圖（桌機/平板模式）`);
        } else if (calendarSection && !isDesktop) {
            calendarSection.style.display = 'none';
            console.log('📱 手機設備：日曆網格視圖不可用，僅顯示卡片列表');
        }
    }
    
    // 更新今日課程區塊
    renderTodayEvents();
}
```

**功能說明：**
- 統一管理 Special Events Manager 的狀態更新
- 根據設備類型自動調整日曆網格顯示
- 自動更新今日課程區塊
- 支援響應式設計

### 4. ✅ 更新視窗大小監聽器（第 9803-9814 行）

```javascript
// 🖥️ 監聽視窗大小變化，動態調整日曆視圖顯示

    clearTimeout(resizeTimeout);
    resizeTimeout = setTimeout(() => {
        // 根據當前視圖更新 Special Events Manager 顯示
```
