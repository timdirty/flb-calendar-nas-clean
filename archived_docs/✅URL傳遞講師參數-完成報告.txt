function goToLearningRecordUpload(eventId, date, time, instructor) {
    // 在 URL 中傳遞講師名稱
    let url = `/learning-record-upload.html?eventId=${...}&date=${...}&time=${...}`;
    if (instructor) {
        url += `&instructor=${encodeURIComponent(instructor)}`;
    }
    window.location.href = url;
}

⚠️ 潛在問題與解決方案：

問題 1: instructor 參數是空值
原因：課程卡片沒有 data-instructor 屬性
解決：系統會退回使用 localStorage

問題 2: 講師名稱不匹配
原因：課程的 instructor 與 teacher_data 的 name 不一致
解決：檢查大小寫，系統會轉為大寫比對

問題 3: 找不到任何課程
原因：該日期該講師沒有已結束的課程
解決：調整日期篩選或狀態篩選
